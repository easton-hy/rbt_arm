%%
set(groot,'defaulttextinterpreter','latex');
set(groot, 'defaultAxesTickLabelInterpreter','latex');
set(groot, 'defaultLegendInterpreter','latex');
%%
q=10*[0.3 -0.2 0.3 -0.2 0.3 -0.2]';
q_dot=10*[0.3 0.3 0.3 -0.3 0.3 -0.3]';
%%
T=0:0.01:15;
dt=0.02;
X=[];
kp=10; kd=2.0;

%%
for i=1:length(T)
    t=T(i);
    qd = zeros(6,1);
    qd_dot =zeros(6,1);
    qd_ddot = zeros(6,1);

    X=[X,[q;qd;q_dot]];

    V=qd_ddot-kp*(q-qd)-kd*(q_dot-qd_dot);
    %% control
    D=1.0*inertiaMatrix(q);
    g=1.0*gravityVector(q);
    U=D*V+g;

    %% dynamcis 
    q_ddot = inertiaMatrix(q)\(U -gravityVector(q)-cor_centriTerms(q,q_dot));
    q_dot=q_ddot*dt+q_dot;
    q=q_dot*dt+q;

    %X=[X,[q;qd;q_dot]];
end

% %%
figure
subplot(6,1,1)
plot(X(1,:))
hold on

ylabel('$q_1$ (rad)')
